# ---------------------------------------------------------------------------
# Qt Quick Test suite: QML components and singletons.
#
# The MSYS2/Windows Qt build does not ship the `qt_add_qml_test` helper, so
# the tests are executed with the `qmltestrunner` binary directly. The
# application module is imported from the build tree, where CMake generates
# the `qmldir` next to the compiled QML sources.
# ---------------------------------------------------------------------------
find_program(QMLTESTRUNNER_EXECUTABLE
    NAMES qmltestrunner qmltestrunner.exe
    HINTS ${Qt6_DIR}/../../../bin
    REQUIRED
)

# The `qmldir` generated by Qt prefers the compiled resources, which are not
# available in the standalone test runner. A disk based module descriptor is
# generated instead so the tests can import the very same QML sources.
set(QML_TEST_IMPORT_ROOT ${CMAKE_CURRENT_BINARY_DIR}/imports)
set(QML_TEST_MODULE_DIR ${QML_TEST_IMPORT_ROOT}/DevDialogue/App)
file(MAKE_DIRECTORY ${QML_TEST_MODULE_DIR})

set(QML_TEST_QMLDIR "module DevDialogue.App\n")
foreach(QML_FILE ${QML_MODULE_FILES})
    get_filename_component(QML_FILE_NAME ${QML_FILE} NAME)
    string(REGEX REPLACE "\\.qml$" "" QML_TYPE_NAME ${QML_FILE_NAME})
    # The QML engine expects a URL, a plain Windows path would be treated as
    # a network location.
    string(REPLACE "\\" "/" QML_FILE_PATH "${CMAKE_SOURCE_DIR}/${QML_FILE}")
    if(QML_FILE IN_LIST QML_SINGLETONS)
        set(QML_TEST_QMLDIR "${QML_TEST_QMLDIR}singleton ${QML_TYPE_NAME} 1.0 file:///${QML_FILE_PATH}\n")
    else()
        set(QML_TEST_QMLDIR "${QML_TEST_QMLDIR}${QML_TYPE_NAME} 1.0 file:///${QML_FILE_PATH}\n")
    endif()
endforeach()
file(WRITE ${QML_TEST_MODULE_DIR}/qmldir "${QML_TEST_QMLDIR}")

add_test(NAME QmlTests
    COMMAND ${QMLTESTRUNNER_EXECUTABLE}
        -import ${QML_TEST_IMPORT_ROOT}
        -input ${CMAKE_CURRENT_SOURCE_DIR}/qml
        -o -,txt
)

set_tests_properties(QmlTests PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
)
